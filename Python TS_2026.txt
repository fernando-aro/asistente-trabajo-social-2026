import streamlit as st
from openai import OpenAI
import os


# 1. Configuración de la interfaz adaptativa para web y móviles
st.set_page_config(
    page_title="Asistente Democracia Participativa - TS 2026", 
    page_icon="⚖️", 
    layout="centered"
)


st.title("🤖 Asistente Virtual: Ponencia Trabajo Social 2026")
st.subheader("Consultas sobre la Teoría de Max-Neef, Ley 8364 y la propuesta de Democracia Participativa")


# 2. Conexión segura con la API Key de OpenAI desde los Secrets de Streamlit
# Esto evita tener que poner la clave en el código o digitarla en pantalla
if "OPENAI_API_KEY" in st.secrets:
    api_key = st.secrets["OPENAI_API_KEY"]
elif "OPENAI_API_KEY" in os.environ:
    api_key = os.environ["OPENAI_API_KEY"]
else:
    st.error("⚠️ No se encontró la API Key de OpenAI. Configúrala en los Secrets de Streamlit Community Cloud.")
    st.stop()


client = OpenAI(api_key=api_key)


# 3. Función optimizada para leer el archivo de contexto externo de forma automática
@st.cache_data
def cargar_contexto_documentos(nombre_archivo="documentos_contexto.txt"):
    """
    Lee de forma automática las normativas, propuestas, referencias bibliográficas 
    y el análisis del IMAS/MIDEPLAN guardados en el archivo externo.
    """
    if not os.path.exists(nombre_archivo):
        # Si el archivo no existe, crea una base para evitar que la aplicación falle
        with open(nombre_archivo, "w", encoding="utf-8") as f:
            f.write("CONTEXTO DE LA PONENCIA:\n(Por favor, pega aquí el contenido de tus propuestas y normativas).")
    
    with open(nombre_archivo, "r", encoding="utf-8") as f:
        return f.read()


# Cargar el texto de los documentos aportados
CONTEXTO_INYECTADO = cargar_contexto_documentos()


# 4. Inicialización del historial de chat en la sesión de Streamlit
if "messages" not in st.session_state:
    st.session_state["messages"] = [
        {
            "role": "system", 
            "content": (
                "Eres un asistente académico experto en Trabajo Social, Gestión Pública y Derechos Humanos. "
                "Responde las dudas de estudiantes y profesionales basándote estrictamente en el siguiente contexto legal, "
                f"teórico y bibliográfico inyectado desde tus documentos oficiales:\n\n{CONTEXTO_INYECTADO}"
            )
        },
        {
            "role": "assistant", 
            "content": (
                "¡Hola! El bot asistente está listo para responder consultas basadas en tus propuestas, las normativas "
                "internacionales de la OGP, OCDE, CEPAL, el marco del Artículo 9 constitucional de Costa Rica, la Ley 8364, "
                "así como la crítica epistemológica al MIDEPLAN/IMAS utilizando a Max-Neef. ¿Qué aspecto deseas consultar en tiempo real?"
            )
        }
    ]


# 5. Renderizar el historial de conversación en la interfaz
for msg in st.session_state.messages:
    if msg["role"] != "system":
        with st.chat_message(msg["role"]):
            st.write(msg["content"])


# 6. Captura de la interacción del usuario
if user_query := st.chat_input("Escribe tu consulta académica o profesional aquí..."):
    # Guardar y mostrar el mensaje del usuario
    st.session_state.messages.append({"role": "user", "content": user_query})
    with st.chat_message("user"):
        st.write(user_query)
        
    # Consultar a la API de OpenAI (utilizando GPT-4o para alta precisión analítica)
    with st.chat_message("assistant"):
        response_placeholder = st.empty()
        try:
            response = client.chat.completions.create(
                model="gpt-4o",  
                messages=st.session_state.messages,
                temperature=0.2  # Temperatura baja para evitar alucinaciones y mantener fidelidad al documento
            )
            answer = response.choices[0].message.content
            response_placeholder.write(answer)
            # Guardar la respuesta en el historial
            st.session_state.messages.append({"role": "assistant", "content": answer})
        except Exception as e:
            st.error(f"Ocurrió un error al procesar la respuesta: {e}")
